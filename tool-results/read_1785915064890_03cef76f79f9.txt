   246→   246→function splitOnce(s: string, sep: string): [string, string] {

   419→   419→    if (!toast) return;

   429→   429→    toastTimer = setTimeout(() => {
   430→   430→      toast.classList.remove('show');
   431→   431→      toastShowing = false;
   432→   432→      if (toastQueue.length > 0) {
   433→   433→        const next = toastQueue.shift()!;
   434→   434→        setTimeout(() => showToast(next.msg, next.duration), 200);

   436→   436→    }, duration);

   579→   579→        const ns: SectionStyle = {};

   581→   581→        if (old.italic === 'italic' || old.fontStyle === 'italic') ns.fontStyle = 'italic';
   582→   582→        const deco = (old.textDecoration as string) || '';
   583→   583→        const parts: string[] = [];
   584→   584→        if (old.underline || deco.includes('underline')) parts.push('underline');
   585→   585→        if (old.strikethrough || deco.includes('line-through')) parts.push('line-through');
   586→   586→        if (parts.length) ns.textDecoration = parts.join(' ');
   587→   587→        if (old.fontSize) ns.fontSize = Number(old.fontSize);
   588→   588→        migrated.sectionStyles[field] = ns;
   589→   589→      });
   590→   590→    }
   591→   591→    return migrated;
   592→   592→  }

   595→   595→  function pruneOrphanWordStyles(card: Card): void {
   596→   596→    if (!card.wordStyles) return;
   597→   597→    const fieldTexts: Record<string, string> = {
   598→   598→      title: card.title,
   599→   599→      subtitle: card.subtitle,
   600→   600→      text: card.text,
   601→   601→      list: card.listItems,
   602→   602→      footer: card.footer,
   603→   603→      cta: card.cta,
   604→   604→    };
   605→   605→    let changed = false;
   606→   606→    Object.keys(card.wordStyles).forEach((key) => {
   607→   607→      const [kf, word] = splitOnce(key, '::');
   608→   608→      const fieldText = fieldTexts[kf] ?? '';
   609→   609→      // Проверяем, существует ли слово как отдельное слово в актуальном тексте
   610→   610→      if (!word || !containsWholeWord(fieldText, word)) {
   611→   611→        delete card.wordStyles[key];
   612→   612→        changed = true;
   613→   613→      }
   614→   614→    });
   615→   615→    if (changed && activeCardIndexForWord !== null && activeFieldForWord) {
   616→   616→      // Если попап открыт — обновим список стилей
   617→   617→      renderWordStyleList();
   618→   618→    }

   621→   621→  function containsWholeWord(text: string, word: string): boolean {
   622→   622→    if (!word) return false;
   623→   623→    let idx = 0;
   624→   624→    while ((idx = text.indexOf(word, idx)) !== -1) {
   625→   625→      const before = text[idx - 1];
   626→   626→      const after = text[idx + word.length];
   627→   627→      if ((!before || !isWordChar(before)) && (!after || !isWordChar(after))) return true;
   628→   628→      idx += word.length;
   629→   629→    }
   630→   630→    return false;
   631→   631→  }

   633→   633→  /* ---------- Применение стилей слов к тексту (фикс #2: корректная работа с & и спецсимволами) ---------- */
   634→   634→  function applyWordStylesToText(text: string, wordStyles: Record<string, WordStyle>, field: string): string {
   635→   635→    if (!text) return '';
   636→   636→    if (!wordStyles || Object.keys(wordStyles).length === 0) return escapeHtml(text);
   637→   637→
   638→   638→    // Собираем применимые стили: {word, styleStr}
   639→   639→    const applicable: { word: string; styleStr: string }[] = [];
   640→   640→    Object.keys(wordStyles).forEach((key) => {
   641→   641→      const [kf, word] = splitOnce(key, '::');
   642→   642→      const keyField = key.includes('::') ? kf : '*';
   643→   643→      if (keyField !== field && keyField !== '*') return;
   644→   644→      const styles = wordStyles[key];
   645→   645→      let styleStr = '';
   646→   646→      if (styles.fontWeight) styleStr += `font-weight:${styles.fontWeight};`;
   647→   647→      if (styles.fontStyle) styleStr += `font-style:${styles.fontStyle};`;
   648→   648→      if (styles.textDecoration) styleStr += `text-decoration:${styles.textDecoration};`;
   649→   649→      if (styles.fontSize) styleStr += `font-size:${styles.fontSize}px;`;
   650→   650→      if (styles.color) styleStr += `color:${escapeHtml(styles.color)};`;
   651→   651→      if (styleStr && word) applicable.push({ word, styleStr });
   652→   652→    });
   653→   653→
   654→   654→    if (applicable.length === 0) return escapeHtml(text);
   655→   655→
   656→   656→    // Находим все вхождения слов (по индексам, с проверкой границ)
   657→   657→    const ranges: { start: number; end: number; styleStr: string }[] = [];
   658→   658→    applicable.forEach(({ word, styleStr }) => {
   659→   659→      let idx = 0;
   660→   660→      while ((idx = text.indexOf(word, idx)) !== -1) {
   661→   661→        const before = text[idx - 1];
   662→   662→        const after = text[idx + word.length];
   663→   663→        if ((!before || !isWordChar(before)) && (!after || !isWordChar(after))) {
   664→   664→          ranges.push({ start: idx, end: idx + word.length, styleStr });
   665→   665→        }
   666→   666→        idx += word.length;
   667→   667→      }
   668→   668→    });
   669→   669→
   670→   670→    if (ranges.length === 0) return escapeHtml(text);
   671→   671→
   672→   672→    // Сортируем по началу; при пересечении выигрывает более раннее
   673→   673→    ranges.sort((a, b) => a.start - b.start || a.end - b.end);
   674→   674→
   675→   675→    let html = '';
   676→   676→    let pos = 0;
   677→   677→    for (const r of ranges) {
   678→   678→      if (r.start < pos) continue; // пересечение — пропускаем
   679→   679→      html += escapeHtml(text.slice(pos, r.start));
   680→   680→      // фикс #13: маркер стилизованного слова (скрывается при экспорте через .exporting)
   681→   681→      html += `<span class="cc-styled-word" style="${r.styleStr}">${escapeHtml(
   682→   682→        text.slice(r.start, r.end),
   683→   683→      )}</span>`;
   684→   684→      pos = r.end;
   685→   685→    }
   686→   686→    html += escapeHtml(text.slice(pos));
   687→   687→    return html;
   688→   688→  }

   757→   757→  function applyCharLimit(): void {
   758→   758→    if (!editorCardsList) return;
   759→   759→    const limit = charLimitEnabled ? (FORMAT_CHAR_LIMITS[currentFormat] || 0) : 0;
   760→   760→    editorCardsList.querySelectorAll<HTMLInputElement | HTMLTextAreaElement>('input[data-field], textarea[data-field]').forEach((el) => {
   761→   761→      if (limit > 0) {
   762→   762→        el.setAttribute('maxlength', String(limit));
   763→   763→      } else {
   764→   764→        // Восстанавливаем исходный maxlength из EDITOR_FIELDS
   765→   765→        const field = el.dataset.field;
   766→   766→        const f = EDITOR_FIELDS.find((ef) => ef.key === field);
