     1→/**
     2→ * CardCraftApp — orchestrator that wires up all modular classes
     3→ * to replace the 2731-line "God Function" in src/lib/card-constructor.ts.
     4→ *
     5→ * Public API:
     6→ *   initCardCraftApp(root)  — boot the app, returns cleanup function
     7→ *   THEME_GROUPS            — re-export for page.tsx static rendering
     8→ *
     9→ * Architecture:
    10→ *   User event → renderer callback → orchestrator dispatches action / mutates card
    11→ *   → StateManager notifies subscribers → UI controls synced
    12→ *   → PreviewRenderer / EditorRenderer / WordEditorManager update DOM
    13→ *   → StorageManager.save (debounced) + HistoryManager.schedulePush (debounced)
    14→ */

    55→    try {
    56→      return fn();
    57→    } catch (err) {
    58→      console.error('[Cardcraft] Error in ' + label + ':', err);
    59→      return undefined;
    60→    }
    61→  }

    63→  const perfEnabled = typeof performance !== 'undefined' && !!performance.now;
    64→  function perfMark(label: string): () => void {
    65→    if (!perfEnabled) return () => {};
    66→    const start = performance.now();
    67→    return () => {
    68→      const dur = performance.now() - start;
    69→      if (dur > 16) {
    70→        console.warn('[Cardcraft:perf] Slow ' + label + ': ' + dur.toFixed(1) + 'ms');
    71→      }
    72→    };
    73→  }

   201→  function saveCardsToLocalStorage({ silent = false } = {}): void {
   202→    try {
   203→      const state = stateManager.get();
   204→      Storage.save({
   205→        cards: state.cards.list,
   206→        theme: state.settings.theme,
   207→        format: state.settings.format,
   208→        showCardNumbers: state.settings.showCardNumbers,
   209→        showProgressBar: state.settings.showProgressBar,
   210→        progressBarStyle: state.settings.progressBarStyle,
   211→        listStyleType: state.settings.listStyleType,
   212→        gradientAngle: state.settings.gradientAngle,
   213→        charLimitEnabled: state.settings.charLimitEnabled,
   214→      });
   215→      if (!silent) showToast('Карточки успешно сохранены!');
   216→    } catch (e) {
   217→      const err = e as Error;
   218→      if (err.message === 'QuotaExceededError') {
   219→        if (!silent) showToast('Недостаточно места. Удалите старые карточки.');
   220→      } else if (!silent) {
   221→        showToast('Ошибка при сохранении карточек');
   222→      }
   223→    }
   224→  }

   226→  function loadCardsFromLocalStorage(): void {
   227→    const saved = Storage.load();
   228→    if (saved.cards && saved.cards.length) {
   229→      stateManager.setCards(saved.cards);
   230→    }
   231→    if (saved.theme) stateManager.dispatch({ type: 'SET_GLOBAL_THEME', payload: saved.theme });
   232→    if (saved.format) stateManager.dispatch({ type: 'SET_FORMAT', payload: saved.format });
   233→    if (saved.showCardNumbers !== undefined)
   234→      stateManager.dispatch({ type: 'SET_SHOW_CARD_NUMBERS', payload: saved.showCardNumbers });
   235→    if (saved.showProgressBar !== undefined)
   236→      stateManager.dispatch({ type: 'SET_SHOW_PROGRESS_BAR', payload: saved.showProgressBar });
   237→    if (saved.progressBarStyle)
   238→      stateManager.dispatch({ type: 'SET_PROGRESS_BAR_STYLE', payload: saved.progressBarStyle });
   239→    if (saved.listStyleType)
   240→      stateManager.dispatch({ type: 'SET_LIST_STYLE', payload: saved.listStyleType });
   241→    if (saved.gradientAngle !== undefined)
   242→      stateManager.dispatch({ type: 'SET_GRADIENT_ANGLE', payload: saved.gradientAngle });
   243→    if (saved.charLimitEnabled !== undefined)
   244→      stateManager.dispatch({ type: 'SET_CHAR_LIMIT', payload: saved.charLimitEnabled });
   245→  }

   276→  function applyCharLimit(): void {
   277→    if (!editorCardsList) return;
   278→    const settings = stateManager.getSettings();
   279→    const limit = settings.charLimitEnabled ? FORMAT_CHAR_LIMITS[settings.format] || 0 : 0;
   280→    editorCardsList
   281→      .querySelectorAll<HTMLInputElement | HTMLTextAreaElement>('input[data-field], textarea[data-field]')
   282→      .forEach((el) => {
   283→        if (limit > 0) {
   284→          el.setAttribute('maxlength', String(limit));
   285→        } else {
   286→          const field = el.dataset.field;
   287→          const f = EDITOR_FIELDS.find((ef) => ef.key === field);
   288→          if (f) el.setAttribute('maxlength', String(f.maxlength));
   289→        }
   290→      });
   291→    if (charCounter) {
   292→      charCounter.style.display = settings.charLimitEnabled && limit > 0 ? '' : 'none';
   293→    }
   294→  }

   296→  function updateCharCounter(idx: number): void {
   297→    const settings = stateManager.getSettings();
   298→    if (!settings.charLimitEnabled || !charCounterText || !charCounter) return;
   299→    const limit = FORMAT_CHAR_LIMITS[settings.format] || 0;
   300→    if (limit <= 0) {
   301→      charCounter.style.display = 'none';
   302→      return;
   303→    }
   304→    const card = stateManager.getCard(idx);
   305→    if (!card) return;
   306→    const totalChars =
   307→      (card.title?.length || 0) +
   308→      (card.subtitle?.length || 0) +
   309→      (card.text?.length || 0) +
   310→      (card.listItems?.length || 0) +
   311→      (card.footer?.length || 0) +
   312→      (card.cta?.length || 0);
   313→    charCounterText.textContent = `${totalChars} / ${limit}`;
   314→    charCounter.classList.toggle('near-limit', totalChars >= limit * 0.9);
   315→  }

   442→  editorRenderer.onAction((action, data) => {
   443→    if (action === 'input' || action === 'paste') {
   444→      const idx = Number(data.index);
   445→      const field = String(data.field);
   446→      const value = String(data.value);
   447→      const card = stateManager.getCard(idx);
   448→      if (!card) return;
   449→      // Mutate card field directly (preserves O(1) update — no dispatch for typing)
   450→      (card as unknown as Record<string, unknown>)[field] = value;
   451→      const changed = pruneOrphanWordStyles(card);
   452→      previewRenderer.updateCardField(card, field, idx);
   453→      updateCharCounter(idx);
   454→      if (changed && activeCardIndexForWord === idx && activeFieldForWord === field) {
   455→        wordEditorManager.renderWordStyleList(card);
   456→      }
   457→      scheduleSave({ silent: true });
   458→      scheduleHistoryPush();
   459→    } else if (action === 'palette') {
   460→      openColorModal(Number(data.index));
   461→    } else if (action === 'delete') {
   462→      deleteCard(Number(data.index));
   463→    } else if (action === 'duplicate') {
   464→      duplicateCard(Number(data.index));
   465→    } else if (action === 'move') {
   466→      moveCard(Number(data.index), Number(data.dir));
   467→    } else if (action === 'focus') {
   468→      updateCharCounter(Number(data.index));
   469→    }
   470→  });

   473→  wordEditorManager.onStyleChange((cardIndex, field, word, styles) => {
   474→    const card = stateManager.getCard(cardIndex);
   475→    if (!card) return;
   476→    const key = `${field}::${word}`;
   477→    const newWordStyles = { ...(card.wordStyles || {}), [key]: styles };
   478→    stateManager.dispatch({
   479→      type: 'SET_CARD_WORD_STYLES',
   480→      payload: { idx: cardIndex, wordStyles: newWordStyles },
   481→    });
   482→    const updated = stateManager.getCard(cardIndex);
   483→    if (updated) {
   484→      previewRenderer.updateCardStyle(updated, field, cardIndex);
   485→      wordEditorManager.renderWordStyleList(updated);
   486→    }
   487→    scheduleSave({ silent: true });
   488→    scheduleHistoryPush();
   489→  });

   491→  wordEditorManager.onRemoveWord((cardIndex, key) => {
   492→    stateManager.dispatch({ type: 'DELETE_CARD_WORD_STYLE', payload: { idx: cardIndex, key } });
   493→    const updated = stateManager.getCard(cardIndex);
   494→    if (!updated) return;
   495→    const [field] = splitOnce(key, '::');
   496→    if (field) previewRenderer.updateCardStyle(updated, field, cardIndex);
   497→    else renderPreview();
   498→    wordEditorManager.renderWordStyleList(updated);
   499→    pushHistory();
   500→    scheduleSave({ silent: true });
   501→  });
   502→
   503→  wordEditorManager.onClear((cardIndex, field, word) => {
   504→    const key = `${field}::${word}`;
   505→    stateManager.dispatch({ type: 'DELETE_CARD_WORD_STYLE', payload: { idx: cardIndex, key } });
   506→    const updated = stateManager.getCard(cardIndex);

   765→  async function copyCardToClipboard(node: HTMLElement): Promise<void> {
   766→    try {
   767→      if (!window.isSecureContext) {
   768→        showToast('Копирование требует HTTPS. Скачиваю PNG вместо копирования…');
   769→        await generateAndDownloadPng(node, 'card-copy.png');
   770→        return;
   771→      }
   772→      const result = await withExportMode(root, node, (n) => Export.copyToClipboard(n, root));
   773→      if (result.success) {
   774→        showToast('Карточка скопирована в буфер!');
   775→        return;
   776→      }
   777→      if (result.fallback) {
   778→        showToast('Буфер обмена недоступен. Скачиваю PNG…');
   779→        await generateAndDownloadPng(node, 'card-copy.png');
   780→      }
   781→    } catch {
   782→      showToast('Копирование не удалось. Скачиваю PNG…');
   783→      try {
   784→        await generateAndDownloadPng(node, 'card-copy.png');
   785→      } catch {
   786→        /* already toasted */
   787→      }
   788→    }
   789→  }

   792→    const cards = stateManager.getCards();

   794→    showToast(`Генерация PNG: 0 из ${total}...`, 60000);
   795→    for (let i = 0; i < total; i++) {
   796→      const node = document.getElementById(`card-node-${cards[i].id}`);
   797→      if (node) {
   798→        try {

   800→        } catch {
   801→          /* ignore — continue batch */
   802→        }
   803→        showToast(`Скачано ${i + 1} из ${total}...`, 60000);
   804→        await new Promise((r) => setTimeout(r, 250));
   805→      }
   806→    }
   807→    showToast(`Готово! Скачано ${total} из ${total} карточек.`);
   808→  }

   811→  function addEl<K extends keyof HTMLElementEventMap>(
   812→    el: EventTarget | null | undefined,
   813→    type: K,
   814→    fn: (e: HTMLElementEventMap[K]) => void,
   815→    opts?: boolean | AddEventListenerOptions,
   816→  ): void {
   817→    if (!el) return;
   818→    const listener = fn as EventListener;
   819→    el.addEventListener(type, listener, opts);
   820→    elementListeners.push({ el, type: type as string, fn: listener, opts });
   821→  }
   822→
   823→  function addDoc<K extends keyof DocumentEventMap>(
   824→    type: K,
   825→    fn: (e: DocumentEventMap[K]) => void,
   826→  ): void {
   827→    const listener = fn as EventListener;
   828→    document.addEventListener(type, listener);
   829→    docListeners.push({ type: type as string, fn: listener });
   830→  }
