   307→   307→      (card.title?.length || 0) +
   308→   308→      (card.subtitle?.length || 0) +
   309→   309→      (card.text?.length || 0) +
   310→   310→      (card.listItems?.length || 0) +
   311→   311→      (card.footer?.length || 0) +
   312→   312→      (card.cta?.length || 0);
   313→   313→    charCounterText.textContent = `${totalChars} / ${limit}`;
   314→   314→    charCounter.classList.toggle('near-limit', totalChars >= limit * 0.9);
   315→   315→  }

   787→   787→      }

   811→   811→  function addEl<K extends keyof HTMLElementEventMap>(
   812→   812→    el: EventTarget | null | undefined,
   813→   813→    type: K,
   814→   814→    fn: (e: HTMLElementEventMap[K]) => void,
   815→   815→    opts?: boolean | AddEventListenerOptions,
   816→   816→  ): void {
   817→   817→    if (!el) return;
   818→   818→    const listener = fn as EventListener;
   819→   819→    el.addEventListener(type, listener, opts);
   820→   820→    elementListeners.push({ el, type: type as string, fn: listener, opts });
   821→   821→  }

   828→   828→    document.addEventListener(type, listener);
   829→   829→    docListeners.push({ type: type as string, fn: listener });
   830→   830→  }
