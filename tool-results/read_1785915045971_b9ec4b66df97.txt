   225→// фикс #5: коллизия идентификаторов — используем crypto.randomUUID()
   226→function generateId(): string {
   227→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) {
   228→    try {
   229→      return crypto.randomUUID();
   230→    } catch {
   231→      /* fall through */
   232→    }
   233→  }
   234→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 11);
   235→}

   237→function escapeHtml(str: string): string {
   238→  return (str || '')
   239→    .replace(/&/g, '&amp;')
   240→    .replace(/</g, '&lt;')
   241→    .replace(/>/g, '&gt;')
   242→    .replace(/"/g, '&quot;')
   243→    .replace(/'/g, '&#039;');
   244→}

   246→function splitOnce(s: string, sep: string): [string, string] {
   247→  const i = s.indexOf(sep);
   248→  if (i === -1) return [s, ''];
   249→  return [s.slice(0, i), s.slice(i + sep.length)];
   250→}

   285→    if (!perfEnabled) return () => {};
   286→    const start = performance.now();
   287→    return () => {
   288→      const dur = performance.now() - start;
   289→      perfCounts[label] = (perfCounts[label] || 0) + 1;
   290→      perfTimes[label] = (perfTimes[label] || 0) + dur;

   292→        // Логируем только медленные вызовы (> 1 кадр)
   293→        console.warn('[Cardcraft:perf] Slow ' + label + ': ' + dur.toFixed(1) + 'ms');
   294→      }
   295→    };
   296→  }

   300→      const count = perfCounts[label];

   418→  function showToast(msg: string, duration = 2500): void {
   419→    if (!toast) return;
   420→    // Если текущий toast — прогресс пакетного скачивания (длинный), заменяем сразу
   421→    if (toastShowing && duration < 10000) {
   422→      toastQueue.push({ msg, duration });
   423→      return;
   424→    }
   425→    toast.textContent = msg;
   426→    toast.classList.add('show');
   427→    toastShowing = true;
   428→    if (toastTimer) clearTimeout(toastTimer);
   429→    toastTimer = setTimeout(() => {
   430→      toast.classList.remove('show');
   431→      toastShowing = false;
   432→      if (toastQueue.length > 0) {
   433→        const next = toastQueue.shift()!;
   434→        setTimeout(() => showToast(next.msg, next.duration), 200);

   436→    }, duration);

   445→  function saveCardsToLocalStorage({ silent = false } = {}): void {
   446→    try {
   447→      const cardsToSave = cards.map((c) => {
   448→        const cleaned: Record<string, unknown> = { ...c };
   449→        if (cleaned.wordStyles && Object.keys(cleaned.wordStyles as object).length === 0)
   450→          delete cleaned.wordStyles;
   451→        if (cleaned.sectionStyles && Object.keys(cleaned.sectionStyles as object).length === 0)
   452→          delete cleaned.sectionStyles;
   453→        if (cleaned.colors && Object.keys(cleaned.colors as object).length === 0)
   454→          delete cleaned.colors;
   455→        if (!cleaned.theme) delete cleaned.theme;
   456→        return cleaned;
   457→      });
   458→      const dataStr = JSON.stringify(cardsToSave);
   459→      const dataSize = new Blob([dataStr]).size;
   460→      try {
   461→        localStorage.setItem('flashcard-cards', dataStr);
   462→        localStorage.setItem('flashcard-theme', currentTheme);
   463→        localStorage.setItem('flashcard-format', currentFormat);
   464→        localStorage.setItem('flashcard-show-numbers', String(showCardNumbers));
   465→        localStorage.setItem('flashcard-show-progress', String(showProgressBar));
   466→        localStorage.setItem('flashcard-progress-style', progressBarStyle);
   467→        localStorage.setItem('flashcard-list-style', listStyleType);
   468→        localStorage.setItem('flashcard-gradient-angle', String(gradientAngle));
   469→        localStorage.setItem('flashcard-char-limit', String(charLimitEnabled));
   470→        if (!silent) showToast('Карточки успешно сохранены!');
   471→      } catch (quotaError) {
   472→        const err = quotaError as Error;
   473→        if (err.name === 'QuotaExceededError') {
   474→          if (!silent)
   475→            showToast(
   476→              `Недостаточно места (${(dataSize / 1024).toFixed(2)} KB). Удалите старые карточки.`,
   477→            );
   478→        } else {
   479→          throw quotaError;
   480→        }
   481→      }
   482→    } catch {
   483→      if (!silent) showToast('Ошибка при сохранении карточек');
   484→    }
   485→  }

   551→  function migrateCard(card: Partial<Card>): Card {
   552→    const migrated: Card = {
   553→      id: card.id || generateId(),
   554→      title: card.title || '',
   555→      subtitle: card.subtitle || '',
   556→      text: card.text || '',
   557→      listItems: card.listItems || '',
   558→      footer: card.footer || '',
   559→      cta: card.cta || '',
   560→      colors: card.colors || {},
   561→      wordStyles: {},
   562→      sectionStyles: card.sectionStyles || {},
   563→      theme: card.theme,
   564→    };
   565→    // Миграция wordStyles: старые ключи (без ::) -> новые (с field::), по умолчанию title
   566→    if (card.wordStyles) {
   567→      Object.keys(card.wordStyles).forEach((key) => {
   568→        if (key.includes('::')) {
   569→          migrated.wordStyles[key] = card.wordStyles![key];
   570→        } else {
   571→          migrated.wordStyles[`title::${key}`] = card.wordStyles![key];
   572→        }
   573→      });
   574→    }
   575→    // Миграция sectionStyles
   576→    if (card.sectionStyles) {
   577→      Object.keys(card.sectionStyles).forEach((field) => {
   578→        const old = card.sectionStyles![field] as Record<string, unknown> & SectionStyle;
   579→        const ns: SectionStyle = {};
   580→        if (old.bold === 'bold' || old.fontWeight === 'bold') ns.fontWeight = 'bold';
   581→        if (old.italic === 'italic' || old.fontStyle === 'italic') ns.fontStyle = 'italic';
   582→        const deco = (old.textDecoration as string) || '';
   583→        const parts: string[] = [];
   584→        if (old.underline || deco.includes('underline')) parts.push('underline');
   585→        if (old.strikethrough || deco.includes('line-through')) parts.push('line-through');
   586→        if (parts.length) ns.textDecoration = parts.join(' ');
   587→        if (old.fontSize) ns.fontSize = Number(old.fontSize);
   588→        migrated.sectionStyles[field] = ns;
   589→      });
   590→    }
   591→    return migrated;
   592→  }

   595→  function pruneOrphanWordStyles(card: Card): void {
   596→    if (!card.wordStyles) return;
   597→    const fieldTexts: Record<string, string> = {
   598→      title: card.title,
   599→      subtitle: card.subtitle,
   600→      text: card.text,
   601→      list: card.listItems,
   602→      footer: card.footer,
   603→      cta: card.cta,
   604→    };
   605→    let changed = false;
   606→    Object.keys(card.wordStyles).forEach((key) => {
   607→      const [kf, word] = splitOnce(key, '::');
   608→      const fieldText = fieldTexts[kf] ?? '';
   609→      // Проверяем, существует ли слово как отдельное слово в актуальном тексте
   610→      if (!word || !containsWholeWord(fieldText, word)) {
   611→        delete card.wordStyles[key];
   612→        changed = true;
   613→      }
   614→    });
   615→    if (changed && activeCardIndexForWord !== null && activeFieldForWord) {
   616→      // Если попап открыт — обновим список стилей
   617→      renderWordStyleList();
   618→    }

   621→  function containsWholeWord(text: string, word: string): boolean {
   622→    if (!word) return false;
   623→    let idx = 0;
   624→    while ((idx = text.indexOf(word, idx)) !== -1) {
   625→      const before = text[idx - 1];
   626→      const after = text[idx + word.length];
   627→      if ((!before || !isWordChar(before)) && (!after || !isWordChar(after))) return true;
   628→      idx += word.length;
   629→    }
   630→    return false;
   631→  }

   633→  /* ---------- Применение стилей слов к тексту (фикс #2: корректная работа с & и спецсимволами) ---------- */
   634→  function applyWordStylesToText(text: string, wordStyles: Record<string, WordStyle>, field: string): string {
   635→    if (!text) return '';
   636→    if (!wordStyles || Object.keys(wordStyles).length === 0) return escapeHtml(text);
   637→
   638→    // Собираем применимые стили: {word, styleStr}
   639→    const applicable: { word: string; styleStr: string }[] = [];
   640→    Object.keys(wordStyles).forEach((key) => {
   641→      const [kf, word] = splitOnce(key, '::');
   642→      const keyField = key.includes('::') ? kf : '*';
   643→      if (keyField !== field && keyField !== '*') return;
   644→      const styles = wordStyles[key];
   645→      let styleStr = '';
   646→      if (styles.fontWeight) styleStr += `font-weight:${styles.fontWeight};`;
   647→      if (styles.fontStyle) styleStr += `font-style:${styles.fontStyle};`;
   648→      if (styles.textDecoration) styleStr += `text-decoration:${styles.textDecoration};`;
   649→      if (styles.fontSize) styleStr += `font-size:${styles.fontSize}px;`;
   650→      if (styles.color) styleStr += `color:${escapeHtml(styles.color)};`;
   651→      if (styleStr && word) applicable.push({ word, styleStr });
   652→    });
   653→
   654→    if (applicable.length === 0) return escapeHtml(text);
   655→
   656→    // Находим все вхождения слов (по индексам, с проверкой границ)
   657→    const ranges: { start: number; end: number; styleStr: string }[] = [];
   658→    applicable.forEach(({ word, styleStr }) => {
   659→      let idx = 0;
   660→      while ((idx = text.indexOf(word, idx)) !== -1) {
   661→        const before = text[idx - 1];
   662→        const after = text[idx + word.length];
   663→        if ((!before || !isWordChar(before)) && (!after || !isWordChar(after))) {
   664→          ranges.push({ start: idx, end: idx + word.length, styleStr });
   665→        }
   666→        idx += word.length;
   667→      }
   668→    });
   669→
   670→    if (ranges.length === 0) return escapeHtml(text);
   671→
   672→    // Сортируем по началу; при пересечении выигрывает более раннее
   673→    ranges.sort((a, b) => a.start - b.start || a.end - b.end);
   674→
   675→    let html = '';
   676→    let pos = 0;
   677→    for (const r of ranges) {
   678→      if (r.start < pos) continue; // пересечение — пропускаем
   679→      html += escapeHtml(text.slice(pos, r.start));
   680→      // фикс #13: маркер стилизованного слова (скрывается при экспорте через .exporting)
   681→      html += `<span class="cc-styled-word" style="${r.styleStr}">${escapeHtml(
   682→        text.slice(r.start, r.end),
   683→      )}</span>`;
   684→      pos = r.end;
   685→    }
   686→    html += escapeHtml(text.slice(pos));
   687→    return html;
   688→  }

   969→  // Поддерживает: обновление in-place, создание (empty→content), удаление (content→empty).
   970→  function updatePreviewField(cardIndex: number, field: string): void {
   971→    const end = perfMark('updatePreviewField');
   972→    try {
   973→      const card = cards[cardIndex];
   974→      if (!card || !cardsArea) {
   975→        renderPreview();
   976→        return;
   977→      }
   978→      const cardNode = document.getElementById(`card-node-${card.id}`);
   979→      if (!cardNode) {
   980→        renderPreview();
   981→        return;
   982→      }
   983→
   984→      const value = (card[field as keyof Card] as string) || '';
   985→
   986→      // listItems → list: специальная обработка (несколько <li>)
   987→      if (field === 'listItems') {
   988→        updatePreviewList(cardNode, card, cardIndex);
   989→        updateEmptyHint(cardNode, card);
   990→        return;
   991→      }
   992→
   993→      // БАГ#4: listNumber/listNumBg/listNumBorder/listNumSize — независимые параметры нумерации
   994→      // Обновляем CSS-переменные на существующих .card-list-num элементах
   995→      if (field === 'listNumber' || field === 'listNumBg' || field === 'listNumBorder' || field === 'listNumSize') {
   996→        const nums = cardNode.querySelectorAll<HTMLElement>('.card-list-num');
   997→        const c = card.colors || {};
   998→        nums.forEach((num) => {
   999→          // Цвет цифры
  1000→          if (c.listNumber) num.style.setProperty('--num-color', c.listNumber);
  1001→          else num.style.removeProperty('--num-color');
  1002→          // Цвет фона фигуры
  1003→          if (c.listNumBg) num.style.setProperty('--num-bg', c.listNumBg);
  1004→          else num.style.removeProperty('--num-bg');
  1005→          // Цвет рамки фигуры
  1006→          if (c.listNumBorder) num.style.setProperty('--num-border', c.listNumBorder);
  1007→          else num.style.removeProperty('--num-border');
  1008→          // Размер фигуры
  1009→          if (c.listNumSize) num.style.setProperty('--num-size', `${c.listNumSize}px`);
  1010→          else num.style.removeProperty('--num-size');
  1011→        });
  1012→        return;
  1013→      }
  1014→
  1015→      const el = cardNode.querySelector<HTMLElement>(`[data-field="${field}"]`);
  1016→
  1017→      // Случай 1: есть контент, элемент существует → обновляем in-place
  1018→      if (value && el) {
  1019→        const styled = applyWordStylesToText(value, card.wordStyles, field);
  1020→        el.innerHTML = field === 'subtitle' || field === 'text' ? styled.replace(/\n/g, '<br>') : styled;
  1021→        updateEmptyHint(cardNode, card);
  1022→        return;
  1023→      }
  1024→
  1025→      // Случай 2: есть контент, элемент НЕ существует → создаём и вставляем
  1026→      if (value && !el) {
  1027→        createFieldElement(cardNode, card, cardIndex, field, value);
  1028→        updateEmptyHint(cardNode, card);
  1029→        return;
  1030→      }
  1031→
  1032→      // Случай 3: контент пуст, элемент существует → удаляем
  1033→      if (!value && el) {
  1034→        el.remove();
  1035→        updateEmptyHint(cardNode, card);
  1036→        return;
  1037→      }
  1038→
  1039→      // Случай 4: контент пуст, элемент не существует → ничего не делаем
  1040→    } catch (err) {
  1041→      console.error('[Cardcraft] Error in updatePreviewField:', err);
  1042→      renderPreview();
  1043→    } finally {
  1044→      end();
  1045→    }
  1046→  }

  1142→  /* ---------- Создание элемента поля и вставка в правильную позицию ---------- */
  1143→  function createFieldElement(
  1144→    cardNode: HTMLElement,
  1145→    card: Card,
  1146→    cardIndex: number,
  1147→    field: string,
  1148→    value: string,
  1149→  ): void {
  1150→    const cfg = FIELD_CONFIG[field];
  1151→    if (!cfg) {
  1152→      // Неизвестное поле — fallback на полную перестройку
  1153→      renderPreview();
  1154→      return;
  1155→    }
  1156→
  1157→    const containerSel = cfg.container === 'top' ? '.card-top-content' : '.card-bottom-content';
  1158→    const container = cardNode.querySelector<HTMLElement>(containerSel);
  1159→    if (!container) {
  1160→      renderPreview();
  1161→      return;
  1162→    }
  1163→
  1164→    // Создаём элемент
  1165→    const el = document.createElement(cfg.tag);
  1166→    el.className = cfg.cls;
  1167→    el.setAttribute('data-field', field);
  1168→    el.setAttribute('data-index', String(cardIndex));
  1169→
  1170→    // Применяем стили секции
  1171→    const styleStr = buildSectionStyle(card, field);
  1172→    if (styleStr) el.setAttribute('style', styleStr.replace('style="', '').replace(/"$/, ''));
  1173→
  1174→    // Контент
  1175→    const styled = applyWordStylesToText(value, card.wordStyles, field);
  1176→    el.innerHTML = field === 'subtitle' || field === 'text' ? styled.replace(/\n/g, '<br>') : styled;
  1177→
  1178→    // Находим позицию для вставки: перед следующим существующим полем
  1179→    const fieldsInOrder = Object.keys(FIELD_CONFIG)
  1180→      .filter((k) => FIELD_CONFIG[k].container === cfg.container)
  1181→      .sort((a, b) => FIELD_CONFIG[a].order - FIELD_CONFIG[b].order);
  1182→
  1183→    const myOrder = cfg.order;
  1184→    let insertBefore: HTMLElement | null = null;
  1185→    for (const f of fieldsInOrder) {
  1186→      if (FIELD_CONFIG[f].order > myOrder) {
  1187→        const nextEl = container.querySelector<HTMLElement>(`[data-field="${f}"]`);
  1188→        // Для list — специальный случай (data-field="list" на .card-list-text, не на ul)
  1189→        if (f === 'list') {
  1190→          const listEl = container.querySelector<HTMLElement>('.card-list');
  1191→          if (listEl) {
  1192→            insertBefore = listEl;
  1193→            break;
  1194→          }
  1195→        } else if (nextEl) {
  1196→          insertBefore = nextEl;
  1197→          break;
  1198→        }
  1199→      }
  1200→    }
  1201→
  1202→    if (insertBefore) {
  1203→      container.insertBefore(el, insertBefore);
  1204→    } else {
  1205→      container.appendChild(el);
  1206→    }
  1207→  }

  1631→    let isDragging = false;
