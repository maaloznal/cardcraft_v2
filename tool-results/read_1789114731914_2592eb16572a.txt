   107→      sm.dispatch({ type: 'DELETE_CARD', payload: { idx: 0 } });
   108→      expect(sm.getCardCount()).toBe(1);
   109→      expect(sm.get()).toBe(before); // same reference — no state change
   110→    });

   112→      sm.dispatch({ type: 'ADD_CARD' });
   113→      const before = sm.get();

   118→      sm.dispatch({ type: 'ADD_CARD' });
   119→      const before = sm.get();
   120→      sm.dispatch({ type: 'DELETE_CARD', payload: { idx: 999 } });
   121→      expect(sm.get()).toBe(before);
   122→    });
   123→    it('with NaN index and >1 cards: documents actual behavior (removes first card via splice(NaN,1) → splice(0,1))', () => {
   124→      // NOTE: The reducer guard `idx < 0 || idx >= length` does NOT catch NaN
   125→      // (NaN comparisons always return false). The guard at the orchestrator
   126→      // level (CardCraftApp.deleteCard) is what prevents NaN from reaching
   127→      // the reducer. Documenting the reducer's actual (unguarded) behavior
   128→      // so a future tightening of the reducer is intentional, not silent.
   129→      sm.dispatch({ type: 'ADD_CARD' }); // 2 cards
   130→      sm.dispatch({ type: 'ADD_CARD' }); // 3 cards
   131→      const firstId = sm.getCards()[0].id;
   132→      sm.dispatch({ type: 'DELETE_CARD', payload: { idx: NaN } });
   133→      // splice(NaN, 1) is treated as splice(0, 1) → first card removed.
   134→      expect(sm.getCardCount()).toBe(2);
   135→      expect(sm.getCards()[0].id).not.toBe(firstId);
   136→    });
   137→    it('with NaN index and only 1 card: guard `length <= 1` triggers, no change', () => {

   140→      expect(sm.get()).toBe(before);
